"""Capture explicitly requested GitHub read-only metadata; never stores credentials."""
import datetime
import json
from pathlib import Path
import subprocess

OUT=Path(__file__).resolve().parent
ROOT=Path(r'F:\AmazingHand-Z19')
records=[]
def api(name,endpoint):
    proc=subprocess.run(['gh','api',endpoint],cwd=ROOT,capture_output=True,encoding='utf-8')
    records.append({'command':'gh api '+endpoint,'exit_code':proc.returncode,'log':name+'.json',
                    'observed_at_utc':datetime.datetime.now(datetime.timezone.utc).isoformat()})
    (OUT/'live-api-commands.json').write_text(json.dumps(records,indent=2),encoding='utf-8')
    if proc.returncode:
        (OUT/(name+'.error.log')).write_text(proc.stderr,encoding='utf-8')
        raise SystemExit('Read failed: '+endpoint+' exit '+str(proc.returncode))
    value=json.loads(proc.stdout)
    (OUT/(name+'.json')).write_text(json.dumps(value,ensure_ascii=False,indent=2),encoding='utf-8')
    return value
repo=api('repository-final-read','repos/Citlali6/AmazingHand-Z19')
branches=api('branches-final-read','repos/Citlali6/AmazingHand-Z19/branches?per_page=100')
protection=api('protection-final-read','repos/Citlali6/AmazingHand-Z19/branches/main/protection')
rules=api('rulesets-final-read','repos/Citlali6/AmazingHand-Z19/rulesets?includes_parents=true&per_page=100')
prs=api('pulls-final-read','repos/Citlali6/AmazingHand-Z19/pulls?state=all&per_page=100')
issues=api('issues-final-read','repos/Citlali6/AmazingHand-Z19/issues?state=all&per_page=100')
expected=json.loads((ROOT/'docs/migration/source-refs-20261019.json').read_text(encoding='utf-8'))['refs']
actual={'refs/heads/'+b['name']:b['commit']['sha'] for b in branches}
source=dict(line.split()[::-1] for line in (OUT/'source-live-refs.log').read_text(encoding='utf-8').splitlines())
assert len(source)==18
assert all(source[ref]==actual[ref]==sha for ref,sha in expected.items())
archive=ROOT/'docs/migration/upstream_archive_20261019_v2/items'
comparisons=[]
for number in [6,16,17,20,15,18,19]:
    live=api(f'source-live-{number}',f'repos/Tom-mr33/AmazingHand-based-on-FPGA/issues/{number}')
    saved=json.loads((archive/str(number)/'issue.json').read_text(encoding='utf-8'))
    fields=['title','body','state','comments']
    same={field:live.get(field)==saved.get(field) for field in fields}
    live_comments=api(f'source-live-comments-{number}',f'repos/Tom-mr33/AmazingHand-based-on-FPGA/issues/{number}/comments?per_page=100')
    saved_comments=json.loads((archive/str(number)/'comments.json').read_text(encoding='utf-8'))
    same['comment_bodies']=[(c['id'],c.get('body')) for c in live_comments]==[(c['id'],c.get('body')) for c in saved_comments]
    comparisons.append({'number':number,'match':same})
summary={'observed_at_utc':datetime.datetime.now(datetime.timezone.utc).isoformat(),
 'repository':repo['full_name'],'owner':repo['owner']['login'],'permissions':repo['permissions'],
 'default_branch':repo['default_branch'],'fork':repo['fork'],'parent':repo['parent']['full_name'],
 'has_issues':repo['has_issues'],'main':actual['refs/heads/main'],
 'checkpoint':actual['refs/heads/feature/migration-ready-20261019'],
 'historical_live_source_branches_match':18,'all_branch_count':len(branches),
 'native_issues':[i['number'] for i in issues if 'pull_request' not in i],
 'pulls':[{'number':p['number'],'draft':p['draft'],'state':p['state'],'head':p['head']['sha'],'base':p['base']['sha']} for p in prs],
 'protection':protection,'rulesets':rules,'source_issue_comparison':comparisons}
(OUT/'live-state.json').write_text(json.dumps(summary,ensure_ascii=False,indent=2),encoding='utf-8')
print(json.dumps({k:v for k,v in summary.items() if k not in ('protection','source_issue_comparison')},indent=2))
print(json.dumps(comparisons,indent=2))
assert all(all(item['match'].values()) for item in comparisons),'Source discussion changed: reread delta'
