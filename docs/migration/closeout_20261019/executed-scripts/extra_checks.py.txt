"""Bounded PC static audit; no imports of project modules or hardware access."""
import ast
import hashlib
import io
import json
import os
from pathlib import Path
import re
import subprocess
import zipfile

ROOT=Path(r'F:\AmazingHand-Z19')
OUT=Path(__file__).resolve().parent
CP='origin/feature/migration-ready-20261019'
def git(*args,ok=(0,)):
    result=subprocess.run(['git',*args],cwd=ROOT,capture_output=True,env=dict(os.environ,GIT_NO_LAZY_FETCH='1'))
    if result.returncode not in ok:
        raise RuntimeError('git '+str(args)+' exit '+str(result.returncode))
    return result.stdout
def blob(ref,path):
    result=subprocess.run(['git','show',ref+':'+path],cwd=ROOT,capture_output=True,env=dict(os.environ,GIT_NO_LAZY_FETCH='1'))
    return result.stdout if result.returncode==0 else None
result={}
objects=git('rev-list','--objects','--all','--missing=print').decode().splitlines()
missing=[line for line in objects if line.startswith('?')]
assert not missing,missing
assert not (ROOT/'.git/objects/info/alternates').exists()
assert not (ROOT/'.git/shallow').exists()
result['git_object_coverage']={'missing':0,'reachable_objects':len(objects),'alternates':False,'shallow':False,
 'note':'Initially filtered network transfer; all reachable missing blobs hydrated and Git blob hashes verified.'}
with zipfile.ZipFile(ROOT/'deliverables/voice_usb/voice_usb_package_20261018_v2.zip') as z:
    prefix='voice_usb_package_20261018_v2/'
    manifest=json.loads(z.read(prefix+'manifest.json'))
    hashes=manifest['sha256']
    for name,sha in hashes.items():
        assert hashlib.sha256(z.read(prefix+name)).hexdigest()==sha,name
    provenance=json.loads(z.read(prefix+'model_provenance.json'))
    wheel_licenses={}
    for name in z.namelist():
        if name.endswith('.whl'):
            with zipfile.ZipFile(io.BytesIO(z.read(name))) as wheel:
                license_names=[n for n in wheel.namelist() if 'license' in n.lower() or 'copying' in n.lower() or 'notice' in n.lower()]
                wheel_licenses[Path(name).name]=license_names
    result['voice_manifest']={'files_verified':len(hashes),'wheels':len(manifest['wheels']),
                              'model_provenance':provenance,'wheel_license_entries':wheel_licenses,
                              'board_validation':manifest['board_validation']}
paths=git('diff','--name-only','origin/main...'+CP).decode().splitlines()
parsed=[]
for path in paths:
    if path.endswith('.py') and not path.startswith('AmazingHand/'):
        data=blob(CP,path)
        if data is not None:
            ast.parse(data,filename=path)
            parsed.append(path)
result['python_syntax']={'passed':len(parsed),'files':parsed,'method':'ast.parse only; no project imports'}
tree=json.loads((OUT/'tree.json').read_text(encoding='utf-8'))['tree']
blobs=[row for row in tree if row['type']=='blob']
result['size']={'checkpoint_file_count':len(blobs),'checkpoint_file_bytes':sum(row['size'] for row in blobs),
                'largest':sorted(blobs,key=lambda row:row['size'],reverse=True)[:8],
                'above_100_MiB':[row['path'] for row in blobs if row['size']>100*1024*1024]}
ws=(OUT/'diff-check.log').read_text(encoding='utf-8').splitlines()
ws_paths={match.group(1) for line in ws if (match:=re.match(r'^(.+):\d+: (?:trailing whitespace|new blank line at EOF|space before tab)',line))}
result['whitespace']={'paths':sorted(ws_paths),'all_in_preserved_payload':all(p.startswith(('AmazingHand/','docs/migration/upstream_archive_20261019_v2/')) for p in ws_paths)}
patterns={
 'github_token':re.compile(rb'(?:gh[pousr]_[A-Za-z0-9]{30,}|github_pat_[A-Za-z0-9_]{30,})'),
 'private_key':re.compile(rb'-----BEGIN (?:RSA |OPENSSH |EC |DSA )?PRIVATE KEY-----'),
 'aws_access_id':re.compile(rb'AKIA[0-9A-Z]{16}'),
 'openai_key':re.compile(rb'sk-(?:proj-)?[A-Za-z0-9_-]{40,}')}
flags=[]
scanned=0
def scan(name,data):
    global scanned
    if len(data)>12*1024*1024 or b'\0' in data[:4096]:
        return
    scanned+=1
    for kind,pattern in patterns.items():
        if pattern.search(data): flags.append({'path':name,'pattern':kind})
for row in blobs:
    path=row['path']
    if row['size']<=12*1024*1024:
        scan(path,(ROOT/path).read_bytes())
for archive in (ROOT/'deliverables').rglob('*.zip'):
    with zipfile.ZipFile(archive) as z:
        for name in z.namelist():
            info=z.getinfo(name)
            if not info.is_dir() and info.file_size<=12*1024*1024 and not name.endswith('.whl'):
                scan(archive.relative_to(ROOT).as_posix()+'!'+name,z.read(name))
result['sensitive_patterns']={'text_candidates_scanned':scanned,'flags':flags,
 'scope':'High-confidence token/private-key patterns in checkpoint and shipped ZIP text, excluding binary/very large members; not a complete security or license audit.'}
# Compare current archived report/source hashes with the three frozen task branches.
fpga='origin/feature/citlali6-camera-hil-unified-20261018'
prefix='fpga/camera_hil/evidence/cdc-2020.1/'
ledger=json.loads(blob(fpga,prefix+'all-evidence.json'))
for name,item in ledger.items():
    assert hashlib.sha256(blob(fpga,prefix+name)).hexdigest()==item['archived_sha256'],name
vision='origin/feature/citlali6-z19-live-vision-20261018'
deployment=json.loads(blob(vision,'docs/fpga/evidence/live_vision_diagnostic_20261018/deployment.json'))
for path,sha in deployment['source_sha256'].items():
    assert hashlib.sha256(blob(vision,path)).hexdigest()==sha,path
result['task_branch_evidence']={'fpga_archived_reports_verified':len(ledger),'vision_runtime_source_hashes_verified':len(deployment['source_sha256']),
 'scope':'PC hash checks of saved historical evidence; no Vivado run, board probe, or hardware acceptance.'}
conflicts=[]
tmp=OUT/'conflict-inputs'
tmp.mkdir(exist_ok=True)
for label,head in [('host','origin/fix/host-failsafe-20261017'),('fpga',fpga),('vision',vision)]:
    for target in ['origin/main',CP]:
        base=git('merge-base',target,head).decode().strip()
        mine=set(git('diff','--name-only',base,head).decode().splitlines())
        theirs=set(git('diff','--name-only',base,target).decode().splitlines())
        shared=[]
        for index,path in enumerate(sorted(mine&theirs)):
            versions=[blob(target,path),blob(base,path),blob(head,path)]
            if any(data is None for data in (versions[0],versions[2])):
                shared.append({'path':path,'status':'manual_delete_modify_check'})
                continue
            files=[]
            for side,data in enumerate(versions):
                f=tmp/f'{label}-{len(conflicts)}-{index}-{side}.txt'
                f.write_bytes(data or b'')
                files.append(str(f))
            proc=subprocess.run(['git','merge-file','--stdout',*files],cwd=OUT,capture_output=True)
            if proc.returncode<0 or proc.returncode>127:
                raise RuntimeError('Merge-file check failed')
            shared.append({'path':path,'status':'text_conflict' if proc.returncode else 'text_merge_clean',
                           'merge_file_exit_code':proc.returncode})
        conflicts.append({'task':label,'head':git('rev-parse',head).decode().strip(),'target':target,
                          'base':base,'shared_paths':shared,'method':'read-only git merge-file --stdout against temporary copies; no branch integration'})
result['integration_preview']=conflicts
(OUT/'extra-checks.json').write_text(json.dumps(result,ensure_ascii=False,indent=2),encoding='utf-8')
print(json.dumps({k:v for k,v in result.items() if k not in ('voice_manifest','size','integration_preview')},ensure_ascii=True,indent=2))
print(json.dumps({'voice_files':result['voice_manifest']['files_verified'],'wheels':result['voice_manifest']['wheels'],'largest':result['size']['largest'][0]},ensure_ascii=True))
print(json.dumps(conflicts,ensure_ascii=True,indent=2))
assert not flags,'Sensitive pattern findings require manual examination before publication'
