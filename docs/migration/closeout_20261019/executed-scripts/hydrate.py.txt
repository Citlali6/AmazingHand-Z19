"""Complete a filtered clone using verified, resumable HTTPS byte ranges."""
import concurrent.futures
import hashlib
import json
from pathlib import Path
import re
import subprocess
from urllib.parse import quote

ROOT = Path(r"F:\AmazingHand-Z19")
OUT = Path(__file__).resolve().parent
PARTS = OUT / "download-parts"
PARTS.mkdir(exist_ok=True)
REF = "71d8bb14d43c67e63f744f506a972bd4cd769d1a"
tree = json.loads((OUT / "tree.json").read_text(encoding="utf-8"))
by_sha = {item['sha']: item for item in tree['tree'] if item['type'] == 'blob'}
def git(*args):
    return subprocess.check_output(['git', *args], cwd=ROOT, encoding='utf-8').strip()
missing = [line[1:] for line in git('rev-list','--objects','--all','--missing=print').splitlines() if line.startswith('?')]
unknown = [sha for sha in missing if sha not in by_sha]
if unknown:
    raise SystemExit('Missing objects outside checkpoint: '+str(unknown))
tasks = []
CHUNK = 4 * 1024 * 1024
for sha in missing:
    item = by_sha[sha]
    for start in range(0,item['size'],CHUNK):
        tasks.append((sha,item,start,min(start+CHUNK,item['size'])-1))
print(json.dumps(dict(missing=len(missing), bytes=sum(by_sha[sha]['size'] for sha in missing), chunks=len(tasks))),flush=True)

def download(task):
    sha,item,start,end = task
    stem = PARTS / f'{sha}-{start}'
    dest = stem.with_suffix('.bin')
    headers = stem.with_suffix('.headers')
    stderr = stem.with_suffix('.stderr')
    url = 'https://raw.githubusercontent.com/Citlali6/AmazingHand-Z19/'+REF+'/'+quote(item['path'],safe='/')+f'?closeout_range={start}'
    args = ['curl.exe','--silent','--show-error','--http1.1','--proxy','http://127.0.0.1:7897',
            '--connect-timeout','10','--max-time','60','--range',f'{start}-{end}',
            '--dump-header',str(headers),'--output',str(dest),url]
    expected_range=f'content-range: bytes {start}-{end}/{item["size"]}'
    for attempt in range(1,4):
        proc=subprocess.run(args,capture_output=True)
        stderr.write_bytes(proc.stderr)
        header_text=headers.read_text(errors='replace').lower() if headers.exists() else ''
        if proc.returncode == 0 and dest.stat().st_size == end-start+1 and expected_range in header_text:
            return dict(sha=sha,path=item['path'],start=start,end=end,exit_code=0,attempt=attempt,
                        command=subprocess.list2cmdline(args),part_sha256=hashlib.sha256(dest.read_bytes()).hexdigest())
    raise RuntimeError(f'Range download failed: {item["path"]} {start}-{end}, exit {proc.returncode}')

records=[]
with concurrent.futures.ThreadPoolExecutor(max_workers=3) as pool:
    for result in pool.map(download,tasks):
        records.append(result)
        (OUT / 'range-downloads.json').write_text(json.dumps(records,indent=2),encoding='utf-8')
        print(json.dumps({key:result[key] for key in ('path','start','end','attempt','exit_code')}),flush=True)

hydrated=[]
for sha in missing:
    item=by_sha[sha]
    data=b''.join((PARTS/f'{sha}-{start}.bin').read_bytes() for start in range(0,item['size'],CHUNK))
    actual=hashlib.sha1(b'blob '+str(len(data)).encode()+b'\0'+data).hexdigest()
    if len(data)!=item['size'] or actual!=sha:
        raise RuntimeError('Object hash mismatch: '+item['path'])
    branch=git('branch','--show-current')
    if not branch or branch=='main':
        raise RuntimeError('Stop: not on named working branch')
    proc=subprocess.run(['git','hash-object','-w','--stdin'],input=data,cwd=ROOT,capture_output=True)
    if proc.returncode!=0 or proc.stdout.decode().strip()!=sha:
        raise RuntimeError('Object store write failed: '+item['path'])
    hydrated.append(dict(path=item['path'],size=len(data),git_blob_sha=sha,sha256=hashlib.sha256(data).hexdigest()))
    print('Verified and hydrated '+item['path'],flush=True)
(OUT/'hydrated.json').write_text(json.dumps(hydrated,indent=2),encoding='utf-8')
remaining=[line for line in git('rev-list','--objects','--all','--missing=print').splitlines() if line.startswith('?')]
if remaining:
    raise RuntimeError('Still missing objects: '+str(remaining))
print('ALL REACHABLE OBJECTS PRESENT; no shared-object dependency',flush=True)
