"""Read-only migration closeout checks; writes evidence only beside this script."""
from pathlib import Path
import datetime
import hashlib
import json
import subprocess
import sys

ROOT = Path(r"F:\AmazingHand-Z19")
OUT = Path(__file__).resolve().parent
CHECKPOINT = "origin/feature/migration-ready-20261019"
records = []

def run(name, args, expected=(0,)):
    proc = subprocess.run(args, cwd=ROOT, capture_output=True, encoding="utf-8", errors="replace")
    log = OUT / (name + ".log")
    log.write_text(proc.stdout + proc.stderr, encoding="utf-8")
    records.append(dict(name=name, command=subprocess.list2cmdline(args), cwd=str(ROOT),
                        exit_code=proc.returncode, log=log.name,
                        log_sha256=hashlib.sha256(log.read_bytes()).hexdigest(),
                        observed_at_utc=datetime.datetime.now(datetime.timezone.utc).isoformat()))
    (OUT / "commands.json").write_text(json.dumps(records, indent=2), encoding="utf-8")
    print(json.dumps(records[-1]), flush=True)
    if proc.returncode not in expected:
        raise SystemExit(f"Stopped: {name}, exit {proc.returncode}; inspect {log}")
    return proc.stdout

if __name__ == "__main__":
    run("status", ["git", "status", "--short", "--branch"])
    run("branch", ["git", "branch", "--show-current"])
    run("remotes", ["git", "remote", "-v"])
    run("refs", ["git", "for-each-ref", "--format=%(refname) %(objectname)", "refs/remotes/origin/"])
    run("commits", ["git", "log", "--reverse", "--format=%H %s", "origin/main.."+CHECKPOINT])
    run("diff-stat", ["git", "diff", "--stat", "origin/main..."+CHECKPOINT])
    run("diff-names", ["git", "diff", "--name-status", "origin/main..."+CHECKPOINT])
    run("diff-check", ["git", "diff", "--check", "origin/main..."+CHECKPOINT], expected=(0,2))
    run("fsck", ["git", "fsck", "--full"])
    run("assets", [sys.executable, "tools/check_migration.py"])
    run("bundle", ["git", "bundle", "verify", "docs/migration/upstream_archive_20261019_v2/pull-merge-previews.bundle"])
    expected = json.loads((ROOT / "docs/migration/source-refs-20261019.json").read_text(encoding="utf-8"))["refs"]
    remote = json.loads((OUT / "branches.json").read_text(encoding="utf-8"))
    actual = {"refs/heads/"+item["name"]: item["commit"]["sha"] for item in remote}
    rows = []
    for ref, sha in expected.items():
        local = run("ref-"+str(len(rows)), ["git", "rev-parse", "origin/"+ref.removeprefix("refs/heads/")]).strip()
        rows.append(dict(ref=ref, expected=sha, remote=actual.get(ref), fetched=local,
                         match=sha == actual.get(ref) == local))
    result = dict(checkpoint=run("checkpoint", ["git", "rev-parse", CHECKPOINT]).strip(),
                  main=run("main", ["git", "rev-parse", "origin/main"]).strip(),
                  branch_count=len(actual), historical_branch_count=len(rows),
                  all_historical_branches_match=all(x["match"] for x in rows), branches=rows)
    (OUT / "ref-comparison.json").write_text(json.dumps(result, indent=2), encoding="utf-8")
    if not result["all_historical_branches_match"]:
        raise SystemExit("Branch mismatch; preserve state")
    print(json.dumps({k:v for k,v in result.items() if k != "branches"}), flush=True)
