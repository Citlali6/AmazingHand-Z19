"""One-off validation of the authorized documentation and URL correction."""
import ast
import hashlib
import json
from pathlib import Path
import re
import subprocess

ROOT=Path(r'F:\AmazingHand-Z19')
OUT=Path(__file__).resolve().parent
e=ROOT/'docs/migration/closeout_20261019'
paths=['.gitattributes','AGENTS.md','README.md','docs/GIT_WORKFLOW.md','docs/log/2026-10-19.md','docs/migration/README.md','tools/archive_github_source.py']
branch=subprocess.check_output(['git','branch','--show-current'],cwd=ROOT,encoding='utf-8').strip()
assert branch=='docs/citlali6-migration-closeout-20261019'
proc=subprocess.run(['git','diff','--check','--',*paths],cwd=ROOT,capture_output=True)
assert proc.returncode==0,proc.stdout
m=json.loads((e/'evidence-manifest.json').read_text(encoding='utf-8'))
for path,item in m.items():
    assert hashlib.sha256((e/path).read_bytes()).hexdigest()==item['sha256'],path
for path in e.glob('*.json'):
    json.loads(path.read_text(encoding='utf-8'))
code=(ROOT/'tools/archive_github_source.py').read_text(encoding='utf-8')
ast.parse(code)
assert 'https://github.com/Citlali6/AmazingHand-Z19/' in code
assert 'xiaomi-he-only/AmazingHand-Z19' not in code
assert "SOURCE = 'Tom-mr33/AmazingHand-based-on-FPGA'" in code
for path in ['AGENTS.md','README.md','docs/GIT_WORKFLOW.md','docs/migration/README.md']:
    assert 'xiaomi-he-only' not in (ROOT/path).read_text(encoding='utf-8'),path
for link in re.findall(r'\]\(([^)]+)\)',(e/'README.md').read_text(encoding='utf-8')):
    if '://' not in link:
        assert (e/link).exists(),link
assert all(not line.endswith((' ','\t')) for line in (e/'README.md').read_text(encoding='utf-8').splitlines())
historical_changed=subprocess.check_output(['git','diff','--name-only','--','AmazingHand','deliverables','docs/migration/upstream_archive_20261019_v2','docs/migration/model-snapshot.json','docs/migration/source-refs-20261019.json','docs/migration/audit_20261019'],cwd=ROOT,encoding='utf-8')
assert not historical_changed,historical_changed
result={'branch':branch,'source_and_docs_diff_check_exit':0,'raw_evidence_excluded_from_whitespace_check':True,
        'evidence_hashes_verified':len(m),'exporter_ast':'passed','live_addresses':'passed',
        'exporter_source_attribution':'preserved','relative_report_links':'passed','historical_payload_diff':'empty',
        'note':'No archive exporter invocation, credentials access, dependency install, project import, or hardware access.'}
(OUT/'doc-validation.json').write_text(json.dumps(result,indent=2),encoding='utf-8')
print(json.dumps(result,indent=2))
