cmake_minimum_required(VERSION 3.10)
project(ah_ncnn_bridge LANGUAGES CXX)
find_package(ncnn REQUIRED CONFIG)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs)
add_executable(ah_ncnn_bridge main.cpp)
target_compile_features(ah_ncnn_bridge PRIVATE cxx_std_14)
target_compile_options(ah_ncnn_bridge PRIVATE -Wall -Wextra -Werror)
target_include_directories(ah_ncnn_bridge PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(ah_ncnn_bridge PRIVATE ncnn ${OpenCV_LIBS})
